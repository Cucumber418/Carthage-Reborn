crb_hannibal.0 = {
	type = country_event
	placement = root

	title = crb_hannibal.0.t
	desc = crb_hannibal.0.d
	flavor = crb_hannibal.0.f

	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		# 双重校验：只有巴卡派获胜、派系斗争结束且汉诺派未获胜时才能发放宣称
		crb_is_carthage_trigger = yes
		has_variable = crb_barca_victory
		has_variable = crb_hanno_barca_concluded
		NOT = { has_variable = crb_hanno_victory }
		NOT = { has_variable = crb_hannibal_concluded }
	}

	option = {
		name = crb_hannibal.0.a
		default_option = yes
		# 大规模恢复祖先领土宣称会引起列强警惕
		change_infamy = 10
		s:STATE_TRIPOLI = { add_claim = root }
		s:STATE_LIBYA = { add_claim = root }
		s:STATE_ALGIERS = { add_claim = root }
		s:STATE_CONSTANTINE = { add_claim = root }
		s:STATE_ORAN = { add_claim = root }
		s:STATE_SICILY = { add_claim = root }
		s:STATE_BALEARIC_ISLANDS = { add_claim = root }
		s:STATE_MALTA = { add_claim = root }
		s:STATE_CRETE = { add_claim = root }
		s:STATE_CYPRUS = { add_claim = root }
		s:STATE_LOWER_ANDALUSIA = { add_claim = root }
		s:STATE_MURCIA = { add_claim = root }
		s:STATE_VALENCIA = { add_claim = root }
		s:STATE_UPPER_ANDALUSIA = { add_claim = root }
		s:STATE_LEBANON = { add_claim = root }
	}
}
