# 君主制倾向的鼓动者出现
crb_hanno_barca.5 = {
	type = country_event
	placement = root

	title = crb_hanno_barca.5.t
	desc = crb_hanno_barca.5.d
	flavor = crb_hanno_barca.5.f

	event_image = {
		video = "votp_agitator_speaking"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	trigger = {
		# 由日志 je_crb_hanno_barca 每年随机触发
	}

	immediate = {
		create_character = {
			is_agitator = yes
			culture = cu:crb_punic
			religion = rel:crb_tanit
			interest_group = ig_armed_forces
			ideology = ideology:ideology_crb_hannibalist
			traits = {
				demagogue
			}
			save_scope_as = crb_royalist_agitator
			on_created = {
				set_temporary_event_immortality = { DURATION = 3 }
			}
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:crb_royalist_agitator }
	}

	option = {
		name = crb_hanno_barca.5.a
		default_option = yes
	}
}
