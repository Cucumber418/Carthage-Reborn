namespace = crb_flavor

# 迦太基特色事件：每半年有一定概率从中随机抽取一个（见 crb_on_actions.txt）
# 事件本身的 trigger 决定能否被抽中；一次性事件用变量记录

# 比尔萨山的牛皮
crb_flavor.1 = {
	type = country_event
	placement = root

	title = crb_flavor.1.t
	desc = crb_flavor.1.d
	flavor = crb_flavor.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	trigger = {
		# 由 on_action crb_flavor_events 触发
		crb_is_carthage_trigger = yes
		NOT = { has_variable = crb_flavor_1_fired }
		any_scope_state = { state_region = s:STATE_TUNISIA }
	}

	immediate = {
		set_variable = crb_flavor_1_fired
	}

	option = {
		name = crb_flavor.1.a
		default_option = yes
		add_modifier = {
			name = crb_flavor_byrsa_survey
			days = normal_modifier_time
		}
	}

	option = {
		name = crb_flavor.1.b
		add_modifier = {
			name = crb_flavor_byrsa_deeds
			days = normal_modifier_time
		}
	}
}
