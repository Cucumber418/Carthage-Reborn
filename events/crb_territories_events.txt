namespace = crb_territories

crb_territories.1 = {
	type = country_event
	placement = root
	title = crb_territories.1.t
	desc = crb_territories.1.d
	flavor = crb_territories.1.f
	duration = 3
	event_image = { video = "europenorthamerica_capitalists_meeting" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	trigger = { crb_is_carthage_trigger = yes is_at_war = no crb_can_found_mauretania_trigger = yes }
	option = {
		name = crb_territories.1.a
		trigger = { crb_is_carthage_trigger = yes is_at_war = no crb_can_found_mauretania_trigger = yes }
		if = {
			limit = { crb_is_carthage_trigger = yes is_at_war = no crb_can_found_mauretania_trigger = yes }
			change_infamy = 10
			custom_tooltip = crb_found_hidden_province_effect_tt
			hidden_effect = { crb_create_province_CPM_effect = yes }
		}
	}
	option = {
		name = crb_territories.1.b
		default_option = yes
	}
}
