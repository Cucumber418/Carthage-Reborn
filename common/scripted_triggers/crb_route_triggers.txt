crb_barca_route_eligible_trigger = {
	crb_is_carthage_trigger = yes
	has_variable = crb_barca_victory
	has_variable = crb_hanno_barca_concluded
	NOT = { has_variable = crb_hanno_victory }
	NOT = { has_variable = crb_hannibal_concluded }
}

# 已结算奖励、推进目标或结束的确切记录；请求标记不代表事件已经发生。
crb_hannibal_has_settled_history_trigger = {
    OR = {
        has_variable = crb_hannibal_intro_rewards_granted
        has_variable = crb_hannibal_concluded
        has_variable = crb_hannibal_tripoli_done
        has_variable = crb_hannibal_numidia_done
        has_variable = crb_hannibal_syracuse_done
        has_variable = crb_hannibal_islands_done
        has_variable = crb_hannibal_iberia_done
        has_variable = crb_hannibal_new_carthage_done
        has_variable = crb_hannibal_tyre_done
        has_modifier = crb_hannibal_returned
    }
}

crb_hannibal_has_history_trigger = {
    OR = {
        has_variable = crb_hannibal_intro_shown
        crb_hannibal_has_settled_history_trigger = yes
    }
}

# 新日志可能先结算目标进度；明确待办且从未创建开场的请求仍可继续。
crb_hannibal_has_unshown_request_trigger = {
    has_variable = crb_hannibal_intro_pending
    has_journal_entry = je_crb_hannibal_returns
    NOT = { has_variable = crb_hannibal_intro_shown }
    NOT = { has_variable = crb_hannibal_intro_rewards_granted }
    NOT = { has_variable = crb_hannibal_concluded }
    NOT = { has_modifier = crb_hannibal_returned }
}
