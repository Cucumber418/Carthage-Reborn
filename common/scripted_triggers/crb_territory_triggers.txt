# PREV is the French-owned state here. The land-only trigger excludes sea crossings.
crb_borders_french_north_africa_trigger = {
	exists = c:FRA
	c:FRA = {
		any_scope_state = {
			crb_is_north_african_state_trigger = yes
			c:TUN ?= { is_land_adjacent_to_state = prev }
		}
	}
}
# Keep the initial political story available after an early transfer to a subject.
crb_initial_conquests_trigger = {
	OR = {
		any_scope_state = {
			OR = {
				state_region = s:STATE_CONSTANTINE
				state_region = s:STATE_TRIPOLI
			}
		}
		any_subject_or_below = {
			has_variable = crb_carthaginian_subject
			any_scope_state = {
				OR = {
					state_region = s:STATE_CONSTANTINE
					state_region = s:STATE_TRIPOLI
				}
			}
		}
	}
}

crb_france_has_north_african_land_trigger = {
	exists = c:FRA
	c:FRA ?= {
		any_scope_state = { crb_is_north_african_state_trigger = yes }
	}
}
crb_gallic_policy_due_trigger = {
	NOT = { has_modifier = crb_gallic_appeasement }
	NOT = { has_modifier = crb_flavor_gallic_legacy }
	NOT = { has_modifier = crb_gallic_nonintervention }
}
crb_can_choose_gallic_policy_trigger = {
	crb_is_carthage_trigger = yes
	crb_france_has_north_african_land_trigger = yes
	OR = {
		has_variable = crb_gallic_cycle_started
		has_variable = crb_flavor_17_fired # 兼容旧档已经遇到过法国的记录
		crb_borders_french_north_africa_trigger = yes
	}
	crb_gallic_policy_due_trigger = yes
}
crb_can_appease_france_trigger = {
	crb_can_choose_gallic_policy_trigger = yes
	custom_tooltip = {
		text = crb_gallic_appeasement_peace_required_tt
		NOT = { any_enemy_in_diplo_play = { this = c:FRA } }
	}
}
