# European territorial definitions. State regions are disjoint from existing territories.

crb_is_lusitania_state_trigger = {
	OR = {
		state_region = s:STATE_ENTRE_DOURO_E_MINHO
		state_region = s:STATE_BEIRA
		state_region = s:STATE_ESTREMADURA
		state_region = s:STATE_ALENTEJO
	}
}

crb_can_found_lusitania_trigger = {
	crb_is_carthage_trigger = yes
	NOT = { exists = c:XLU }
	any_scope_state = { is_capital = no crb_is_lusitania_state_trigger = yes }
}
