# 汉尼拔再临（巴卡派路线）
je_crb_hannibal_returns = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_crb_carthage

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		c:TUN ?= THIS
	}

	is_shown_when_inactive = {
		crb_is_carthage_trigger = yes
		has_variable = crb_barca_victory
		NOT = { has_variable = crb_hanno_victory }
		has_variable = crb_hanno_barca_concluded
		NOT = { has_variable = crb_hannibal_concluded }
	}

	possible = {
		crb_is_carthage_trigger = yes
		has_variable = crb_barca_victory
		NOT = { has_variable = crb_hanno_victory }
		has_variable = crb_hanno_barca_concluded
		NOT = { has_variable = crb_hannibal_concluded }
	}

	immediate = {
		# 宣称在开场事件的选项中获得，这样玩家可以在选项上看到
		if = {
			limit = {
				NOT = { has_variable = crb_hanno_victory }
				NOT = { has_variable = crb_hannibal_intro_shown }
			}
			set_variable = crb_hannibal_intro_shown
			trigger_event = { id = crb_hannibal.0 popup = yes }
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_variable = crb_hannibal_tripoli_done }
					crb_controls_region_trigger = { REGION = STATE_TRIPOLI }
					crb_controls_region_trigger = { REGION = STATE_LIBYA }
				}
				set_variable = crb_hannibal_tripoli_done
				trigger_event = { id = crb_hannibal.1 popup = yes }
			}
			if = {
				limit = {
					NOT = { has_variable = crb_hannibal_numidia_done }
					crb_controls_region_trigger = { REGION = STATE_ALGIERS }
					crb_controls_region_trigger = { REGION = STATE_CONSTANTINE }
					crb_controls_region_trigger = { REGION = STATE_ORAN }
				}
				set_variable = crb_hannibal_numidia_done
				trigger_event = { id = crb_hannibal.2 popup = yes }
			}
			if = {
				limit = {
					NOT = { has_variable = crb_hannibal_syracuse_done }
					crb_controls_region_trigger = { REGION = STATE_SICILY }
				}
				set_variable = crb_hannibal_syracuse_done
				trigger_event = { id = crb_hannibal.3 popup = yes }
			}
			if = {
				limit = {
					NOT = { has_variable = crb_hannibal_islands_done }
					crb_controls_region_trigger = { REGION = STATE_BALEARIC_ISLANDS }
					crb_controls_region_trigger = { REGION = STATE_MALTA }
					crb_controls_region_trigger = { REGION = STATE_CRETE }
					crb_controls_region_trigger = { REGION = STATE_CYPRUS }
				}
				set_variable = crb_hannibal_islands_done
				trigger_event = { id = crb_hannibal.4 popup = yes }
			}
			if = {
				limit = {
					NOT = { has_variable = crb_hannibal_iberia_done }
					crb_controls_region_trigger = { REGION = STATE_LOWER_ANDALUSIA }
				}
				set_variable = crb_hannibal_iberia_done
				trigger_event = { id = crb_hannibal.5 popup = yes }
			}
			if = {
				limit = {
					NOT = { has_variable = crb_hannibal_new_carthage_done }
					crb_controlled_iberian_states >= 3
				}
				set_variable = crb_hannibal_new_carthage_done
				trigger_event = { id = crb_hannibal.6 popup = yes }
			}
			if = {
				limit = {
					NOT = { has_variable = crb_hannibal_tyre_done }
					OR = {
                        any_scope_state = { state_region = s:STATE_LEBANON }
                        any_subject_or_below = {
                            has_variable = crb_carthaginian_subject
                            any_scope_state = { state_region = s:STATE_LEBANON }
                        }
                    }
				}
				set_variable = crb_hannibal_tyre_done
				trigger_event = { id = crb_hannibal.7 popup = yes }
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = crb_hannibal_tripoli_tt
				has_variable = crb_hannibal_tripoli_done
			}
			custom_tooltip = {
				text = crb_hannibal_numidia_tt
				has_variable = crb_hannibal_numidia_done
			}
			custom_tooltip = {
				text = crb_hannibal_syracuse_tt
				has_variable = crb_hannibal_syracuse_done
			}
			custom_tooltip = {
				text = crb_hannibal_islands_tt
				has_variable = crb_hannibal_islands_done
			}
			custom_tooltip = {
				text = crb_hannibal_iberia_tt
				has_variable = crb_hannibal_iberia_done
			}
			custom_tooltip = {
				text = crb_hannibal_new_carthage_tt
				has_variable = crb_hannibal_new_carthage_done
			}
			custom_tooltip = {
				text = crb_hannibal_tyre_tt
				has_variable = crb_hannibal_tyre_done
			}
		}
		effect = {
			set_variable = crb_hannibal_concluded
			show_as_tooltip = {
				add_modifier = {
					name = crb_hannibal_returned
					days = very_long_modifier_time
				}
			}
			trigger_event = { id = crb_hannibal.8 popup = yes }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { crb_is_carthage_trigger = yes }
		}
		effect = {
			set_variable = crb_hannibal_concluded
		}
	}

	# 防御性校验：若汉诺派胜利标记存在，立即使这条巴卡派专属日志失效
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_variable = crb_hanno_victory
		}
		effect = {
			set_variable = crb_hannibal_concluded
		}
	}

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
