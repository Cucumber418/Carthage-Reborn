# 迦太基文化复兴
je_crb_cultural_revival = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_crb_carthage

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		c:TUN ?= THIS
	}

	is_shown_when_inactive = {
		crb_is_carthage_trigger = yes
		NOT = { has_variable = crb_revival_concluded }
	}

	possible = {
		crb_is_carthage_trigger = yes
		NOT = { has_variable = crb_revival_concluded }
	}

	scripted_button = crb_salt_marsh_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = crb_revival_status_salt_in_progress
				trigger = {
					has_modifier = crb_salt_reclamation
				}
			}
			triggered_desc = {
				desc = crb_revival_status_default
				trigger = {
					always = yes
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			# 复兴种植园
			if = {
				limit = {
					NOT = { has_variable = crb_revival_plantations_done }
					crb_plantation_levels >= 20
				}
				set_variable = crb_revival_plantations_done
				trigger_event = { id = crb_revival.1 popup = yes }
			}
			# 推广多神教
			if = {
				limit = {
					NOT = { has_variable = crb_revival_tanit_done }
					religion_percent_country = {
						target = rel:crb_tanit
						value >= 0.8
					}
				}
				set_variable = crb_revival_tanit_done
				trigger_event = { id = crb_revival.3 popup = yes }
			}
			# 重建港口：研究钢筋混凝土后解锁 U 型港
			if = {
				limit = {
					NOT = { has_variable = crb_cothon_unlocked }
					has_technology_researched = reinforced_concrete
				}
				set_variable = crb_cothon_unlocked
				trigger_event = { id = crb_revival.5 popup = yes }
			}
			if = {
				limit = {
					NOT = { has_variable = crb_revival_harbor_done }
					crb_has_completed_cothon_trigger = yes
				}
				set_variable = crb_revival_harbor_done
				trigger_event = { id = crb_revival.4 popup = yes }
			}
		}
	}

	on_yearly_pulse = {
		effect = {
			# 治理盐碱地：每年计一次，满 10 年完成
			if = {
				limit = { has_modifier = crb_salt_reclamation }
				change_variable = {
					name = crb_salt_years
					add = 1
				}
				if = {
					limit = { var:crb_salt_years >= 10 }
					remove_modifier = crb_salt_reclamation
					set_variable = crb_revival_salt_done
					trigger_event = { id = crb_revival.2 popup = yes }
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = crb_revival_plantations_tt
				has_variable = crb_revival_plantations_done
			}
			custom_tooltip = {
				text = crb_revival_salt_tt
				has_variable = crb_revival_salt_done
			}
			custom_tooltip = {
				text = crb_revival_tanit_tt
				has_variable = crb_revival_tanit_done
			}
			custom_tooltip = {
				text = crb_revival_harbor_tt
				has_variable = crb_revival_harbor_done
			}
		}
		effect = {
			set_variable = crb_revival_concluded
			show_as_tooltip = {
				add_modifier = {
					name = crb_punic_revival
				}
			}
			trigger_event = { id = crb_revival.6 popup = yes }
		}
	}

	# 主流文化不再是布匿
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { crb_is_carthage_trigger = yes }
		}
		effect = {
			set_variable = crb_revival_concluded
			remove_modifier = crb_salt_reclamation
		}
	}

	weight = 900

	should_be_pinned_by_default_uninvolved_or_context = yes
}
