# 两千年的复仇路（巴卡派路线）
# 通过日志里的按钮推进：翻过比利牛斯！、越过高卢！、翻过阿尔卑斯！、复仇开始！
# 复仇完成后日志不关闭，继续用来组建商站、行省
je_crb_revenge_road = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_crb_carthage

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		c:TUN ?= THIS
	}

	is_shown_when_inactive = {
		crb_is_carthage_trigger = yes
		has_variable = crb_hannibal_new_carthage_done
		NOT = { has_variable = crb_revenge_concluded }
	}

	possible = {
		crb_is_carthage_trigger = yes
		has_variable = crb_hannibal_new_carthage_done
		NOT = { has_variable = crb_revenge_concluded }
	}

	immediate = {
		trigger_event = { id = crb_revenge.0 popup = yes }
	}

	scripted_button = crb_revenge_stage1_button
	scripted_button = crb_revenge_stage2_button
	scripted_button = crb_revenge_stage3_button
	scripted_button = crb_revenge_stage4_button

	# 组建商站、行省：日志一直保留，玩家可以随时回来组建
	scripted_button = crb_found_tripoli_post_button
	scripted_button = crb_found_numidia_post_button
	scripted_button = crb_found_iberia_post_button
	scripted_button = crb_found_gaul_post_button
	scripted_button = crb_found_alpine_post_button
	scripted_button = crb_handle_rome_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = crb_revenge_status_5
				trigger = { has_variable = crb_revenge_completed }
			}
			triggered_desc = {
				desc = crb_revenge_status_4
				trigger = { has_variable = crb_revenge_stage4_taken }
			}
			triggered_desc = {
				desc = crb_revenge_status_3
				trigger = { has_variable = crb_revenge_stage3_taken }
			}
			triggered_desc = {
				desc = crb_revenge_status_2
				trigger = { has_variable = crb_revenge_stage2_taken }
			}
			triggered_desc = {
				desc = crb_revenge_status_1
				trigger = { has_variable = crb_revenge_stage1_taken }
			}
			triggered_desc = {
				desc = crb_revenge_status_0
				trigger = { always = yes }
			}
		}
	}

	# 复仇完成：拿下罗马并结束战争
	# 日志不会因此关闭，之后仍可以在这里组建商站、行省
	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_variable = crb_revenge_completed }
					has_variable = crb_revenge_stage4_taken
					crb_controls_region_trigger = { REGION = STATE_LAZIO }
					is_at_war = no
				}
				set_variable = crb_revenge_completed
				trigger_event = { id = crb_revenge.9 popup = yes }
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { crb_is_carthage_trigger = yes }
		}
		effect = {
			set_variable = crb_revenge_concluded
		}
	}

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
