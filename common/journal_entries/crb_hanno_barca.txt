# 汉诺派与巴卡派
# 结构参照原版 je_congress_poland（白党和红党）：一条双向进度条，两派各自的胜利条件
je_crb_hanno_barca = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_crb_carthage

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		country_has_primary_culture = cu:crb_punic
		NOT = { has_variable = crb_hanno_barca_concluded }
	}

	possible = {
		country_has_primary_culture = cu:crb_punic
		NOT = { has_variable = crb_hanno_barca_concluded }
		custom_tooltip = {
			text = crb_hb_claims_taken_tt
			any_scope_state = {
				OR = {
					state_region = s:STATE_TRIPOLI
					state_region = s:STATE_CONSTANTINE
				}
			}
		}
	}

	immediate = {
		ig:ig_landowners ?= { save_scope_as = crb_hb_landowners }
		ig:ig_industrialists ?= { save_scope_as = crb_hb_industrialists }
		ig:ig_devout ?= { save_scope_as = crb_hb_devout }
		ig:ig_armed_forces ?= { save_scope_as = crb_hb_armed_forces }
		ig:ig_petty_bourgeoisie ?= { save_scope_as = crb_hb_petty_bourgeoisie }
		ig:ig_rural_folk ?= { save_scope_as = crb_hb_rural_folk }
		set_variable = {
			name = crb_hb_stalemate_months
			value = 0
		}
		# 凯旋的余威：日志开始后 2 年内巴卡派额外增长
		set_variable = {
			name = crb_hb_triumph
			days = 730
		}
		trigger_event = {
			id = crb_hanno_barca.1
			popup = yes
		}
	}

	scripted_progress_bar = crb_hanno_barca_progress_bar

	scripted_button = crb_hb_concede_senate_button
	scripted_button = crb_hb_concede_assembly_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = crb_hb_status_stalemate
				trigger = {
					scope:journal_entry ?= {
						"scripted_bar_progress(crb_hanno_barca_progress_bar)" >= 30
						"scripted_bar_progress(crb_hanno_barca_progress_bar)" <= 70
					}
				}
			}
			triggered_desc = {
				desc = crb_hb_status_barca_leading
				trigger = {
					scope:journal_entry ?= {
						"scripted_bar_progress(crb_hanno_barca_progress_bar)" > 70
					}
				}
			}
			triggered_desc = {
				desc = crb_hb_status_hanno_leading
				trigger = {
					always = yes
				}
			}
		}
	}

	on_monthly_pulse = {
		# 随机事件：平均每 5 个月左右一个，每个事件 5 年内只出现一次
		random_events = {
			160 = 0
			5 = crb_hanno_barca_random.10
			5 = crb_hanno_barca_random.11
			5 = crb_hanno_barca_random.12
			5 = crb_hanno_barca_random.13
			5 = crb_hanno_barca_random.14
			5 = crb_hanno_barca_random.15
			5 = crb_hanno_barca_random.16
			5 = crb_hanno_barca_random.17
		}
		effect = {
			# 僵持计数：双方力量都在 30% 到 70% 之间时累计，否则清零
			if = {
				limit = {
					scope:journal_entry ?= {
						"scripted_bar_progress(crb_hanno_barca_progress_bar)" >= 30
						"scripted_bar_progress(crb_hanno_barca_progress_bar)" <= 70
					}
				}
				change_variable = {
					name = crb_hb_stalemate_months
					add = 1
				}
			}
			else = {
				set_variable = {
					name = crb_hb_stalemate_months
					value = 0
				}
			}
		}
	}

	on_yearly_pulse = {
		effect = {
			# 巴卡派越强，越容易出现汉尼拔主义者鼓动者（君主制倾向）
			if = {
				limit = {
					NOT = {
						any_scope_character = {
							has_role = character_role_agitator
							has_ideology = ideology:ideology_crb_hannibalist
						}
					}
				}
				random = {
					chance = crb_hb_royalist_agitator_chance
					trigger_event = {
						id = crb_hanno_barca.5
						popup = yes
					}
				}
			}
		}
	}

	# 巴卡派获胜
	outcome = {
		header = crb_hb_barca_victory
		effect_header = crb_hb_barca_victory_effect_header
		color = "red_outcome"
		conclusion_desc = crb_hb_barca_victory_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = crb_hb_barca_victory_bar_tt
				scope:journal_entry ?= {
					"scripted_bar_progress(crb_hanno_barca_progress_bar)" >= 100
				}
			}
		}
		effect = {
			set_variable = crb_hanno_barca_concluded
			crb_barca_victory_effect = yes
			hidden_effect = { crb_hanno_barca_cleanup_effect = yes }
			trigger_event = {
				id = crb_hanno_barca.2
				popup = yes
			}
		}
	}

	# 汉诺派获胜
	outcome = {
		header = crb_hb_hanno_victory
		effect_header = crb_hb_hanno_victory_effect_header
		color = "white_outcome"
		conclusion_desc = crb_hb_hanno_victory_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = crb_hb_hanno_victory_bar_tt
				scope:journal_entry ?= {
					"scripted_bar_progress(crb_hanno_barca_progress_bar)" <= 0
				}
			}
		}
		effect = {
			set_variable = crb_hanno_barca_concluded
			crb_hanno_victory_effect = yes
			hidden_effect = { crb_hanno_barca_cleanup_effect = yes }
			trigger_event = {
				id = crb_hanno_barca.3
				popup = yes
			}
		}
	}

	# 僵持 4 年：弹出事件，由玩家选择倒向哪一派
	outcome = {
		header = crb_hb_stalemate
		effect_header = crb_hb_stalemate_effect_header
		color = "gold_positive"
		conclusion_desc = crb_hb_stalemate_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = crb_hb_stalemate_tt
				has_variable = crb_hb_stalemate_months
				var:crb_hb_stalemate_months >= 48
			}
		}
		effect = {
			set_variable = crb_hanno_barca_concluded
			hidden_effect = { crb_hanno_barca_cleanup_effect = yes }
			trigger_event = {
				id = crb_hanno_barca.4
				popup = yes
			}
		}
	}

	# 主流文化不再是布匿：日志失效
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { country_has_primary_culture = cu:crb_punic }
		}
		effect = {
			set_variable = crb_hanno_barca_concluded
			hidden_effect = { crb_hanno_barca_cleanup_effect = yes }
		}
	}

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
