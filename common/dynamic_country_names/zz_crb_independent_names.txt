# Append independent geographic names without replacing existing subject names.
# INJECT_OR_CREATE also covers IOP/RPR, which have no existing dynamic-name object.
# Source: Modding-Digests/1.12.0/inject_types.md (dynamic_country_names is supported).

INJECT_OR_CREATE:TRP = {
	dynamic_country_name = {
		name = TRP_crb_independent
		adjective = TRP_ADJ
		is_main_tag_only = yes
		priority = 1000
		trigger = {
			scope:actor ?= {
				has_variable = crb_carthaginian_subject
				is_subject = no
			}
		}
	}
}
