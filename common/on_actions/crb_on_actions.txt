# 挂到本体的 on_action 上（同名 on_action 的 on_actions 列表会合并，不会覆盖本体）

on_game_started = {
	on_actions = {
		crb_on_game_started
	}
}

on_monthly_pulse_country = {
	on_actions = {
		crb_monthly_pulse_country
	}
}

# 开局：家园、人口转换、利益集团改名、开局事件
crb_on_game_started = {
	effect = {
		s:STATE_TUNISIA = {
			add_homeland = cu:crb_punic
		}
		c:TUN ?= {
			crb_convert_starting_pops_effect = yes
			crb_update_ig_names_effect = yes
			crb_apply_punic_state_names_effect = yes
			trigger_event = {
				id = crb_carthage.1
				popup = yes
			}
		}
	}
}

# 每月：检查利益集团名称是否需要切换（受认可、国教变化），受认可后移除开局修正
crb_monthly_pulse_country = {
	trigger = {
		exists = c:TUN
		this = c:TUN
	}
	effect = {
		crb_update_ig_names_effect = yes
		# 历史地名：新拥有的地区改名，失去的地区恢复原名
		crb_apply_punic_state_names_effect = yes
		crb_reset_lost_punic_state_names_effect = yes
		# 受认可后移除"迦太基仍未毁灭"
		if = {
			limit = {
				has_modifier = crb_carthago_non_deleta
				NOT = { is_country_type = unrecognized }
			}
			remove_modifier = crb_carthago_non_deleta
		}
	}
}
