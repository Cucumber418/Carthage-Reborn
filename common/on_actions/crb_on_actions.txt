# 挂到本体的 on_action 上（同名 on_action 的 on_actions 列表会合并，不会覆盖本体）

on_game_started = {
	on_actions = {
		crb_on_game_started
	}
}

# 地区易手、新建地区时：在原版动态地名之后套用迦太基历史地名（根作用域：州）
on_state_owner_change = {
	on_actions = {
		crb_on_state_changed
	}
}

on_state_created = {
	on_actions = {
		crb_on_state_changed
	}
}

crb_on_state_changed = {
	effect = {
		crb_assign_punic_name_to_state_effect = yes
	}
}

on_monthly_pulse_country = {
	on_actions = {
		crb_monthly_pulse_country
		crb_monthly_pulse_subject
	}
}

# 开局：家园、人口转换、利益集团改名、开局事件
crb_on_game_started = {
	effect = {
		s:STATE_TUNISIA = {
			add_homeland = cu:crb_punic
		}
		c:TUN ?= {
			crb_convert_starting_pops_effect = yes
			crb_update_ig_names_effect = yes
			crb_apply_punic_state_names_effect = yes
			trigger_event = {
				id = crb_carthage.1
				popup = yes
			}
		}
	}
}

# 每月：检查利益集团名称是否需要切换（受认可、国教变化），受认可后移除开局修正
crb_monthly_pulse_country = {
	trigger = {
		exists = c:TUN
		this = c:TUN
	}
	effect = {
		crb_update_ig_names_effect = yes
		# 历史地名：每月重新套用一次，防止被原版动态地名覆盖
		crb_apply_punic_state_names_effect = yes
		# 第 N 次布匿战争
		crb_punic_war_check_effect = yes
		# 受认可后移除"迦太基仍未毁灭"
		if = {
			limit = {
				has_modifier = crb_carthago_non_deleta
				NOT = { is_country_type = unrecognized }
			}
			remove_modifier = crb_carthago_non_deleta
		}
	}
}

# 每月：迦太基建立的傀儡国、商站也使用历史地名
crb_monthly_pulse_subject = {
	trigger = {
		has_variable = crb_carthaginian_subject
	}
	effect = {
		crb_apply_punic_state_names_effect = yes
	}
}

# 外交博弈开始、有国家加入博弈时：检查是否要开始"第 N 次布匿战争"（根作用域：外交博弈）
on_diplomatic_play_started = {
	on_actions = {
		crb_on_diplo_play_changed
	}
}

on_diplo_play_join_side = {
	on_actions = {
		crb_on_diplo_play_changed
	}
}

crb_on_diplo_play_changed = {
	effect = {
		every_scope_play_involved = {
			limit = { crb_is_carthage_trigger = yes }
			crb_punic_war_check_effect = yes
		}
	}
}
