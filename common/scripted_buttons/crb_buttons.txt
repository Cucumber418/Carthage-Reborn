# 迦太基文化复兴：开始治理盐碱地
crb_salt_marsh_button = {
	name = "crb_salt_marsh_button"
	desc = "crb_salt_marsh_button_desc"

	visible = {
		has_journal_entry = je_crb_cultural_revival
		NOT = { has_variable = crb_revival_salt_done }
		NOT = { has_modifier = crb_salt_reclamation }
	}

	possible = {
		has_technology_researched = intensive_agriculture
		custom_tooltip = {
			text = crb_owns_carthage_tt
			any_scope_state = {
				state_region = s:STATE_TUNISIA
			}
		}
	}

	ai_chance = {
		value = 50
	}

	effect = {
		add_modifier = {
			name = crb_salt_reclamation
			multiplier = money_amount_multiplier_medium
		}
		if = {
			limit = { NOT = { has_variable = crb_salt_years } }
			set_variable = {
				name = crb_salt_years
				value = 0
			}
		}
	}
}

# 第 N 次布匿战争：动员公民
crb_punic_war_mobilize_button = {
	name = "crb_punic_war_mobilize_button"
	desc = "crb_punic_war_mobilize_button_desc"

	visible = {
		has_journal_entry = je_crb_punic_war
	}

	possible = {
		is_at_war = yes
		custom_tooltip = {
			text = crb_punic_war_mobilize_cooldown_tt
			NOT = { has_modifier = crb_punic_war_mobilization }
		}
	}

	ai_chance = {
		value = 50
	}

	effect = {
		every_scope_war = {
			add_war_war_support = {
				target = root
				value = 10
			}
		}
		add_modifier = {
			name = crb_punic_war_mobilization
			multiplier = money_amount_multiplier_small
			days = very_short_modifier_time
		}
	}
}
