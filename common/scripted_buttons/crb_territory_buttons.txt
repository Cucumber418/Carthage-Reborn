crb_found_tripoli_post_button = {
	name = crb_found_tripoli_post_button
	desc = crb_found_tripoli_post_button_desc
	visible = {
		has_journal_entry = je_crb_territories
		crb_can_found_tripoli_trigger = yes
	}
	possible = {
		crb_territory_disposition_ready_trigger = yes
		is_at_war = no
		crb_can_found_tripoli_trigger = yes
	}
	ai_chance = { value = 50 }
	effect = {
		if = {
			limit = { is_at_war = no crb_territory_disposition_ready_trigger = yes crb_can_found_tripoli_trigger = yes }
			trigger_event = { id = crb_revenge.10 popup = yes }
		}
	}
}
