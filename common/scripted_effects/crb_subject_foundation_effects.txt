# New subject country scope; the founder is saved by the territory factory.
crb_initialize_territory_culture_and_laws_effect = {
	add_primary_culture = cu:crb_punic
	activate_law = law_type:law_subjecthood
	if = {
		limit = {
			scope:crb_territory_origin = {
				OR = {
					has_law_or_variant = law_type:law_internal_colonization
					has_law_or_variant = law_type:law_frontier_colonization
					has_law_or_variant = law_type:law_colonial_resettlement
					has_law_or_variant = law_type:law_colonial_exploitation
				}
			}
		}
		activate_law = law_type:law_frontier_colonization
	}
}
