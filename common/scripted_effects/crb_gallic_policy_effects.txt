# 两国的实际停战协议独立存在，法国退出北非时仍按签订期限履行。
crb_stop_gallic_cycle_effect = {
	remove_variable = crb_gallic_cycle_started
	remove_variable = crb_gallic_choice_pending
	remove_variable = crb_flavor_17_fired
	remove_modifier = crb_gallic_appeasement
	remove_modifier = crb_flavor_gallic_legacy
	remove_modifier = crb_gallic_nonintervention
}

# 国家作用域；POLICY 为本轮五年路线。
crb_start_gallic_policy_effect = {
	remove_modifier = crb_gallic_appeasement
	remove_modifier = crb_flavor_gallic_legacy
	remove_modifier = crb_gallic_nonintervention
	remove_variable = crb_gallic_choice_pending
	set_variable = crb_gallic_cycle_started
	set_variable = crb_flavor_17_fired
	add_modifier = { name = $POLICY$ months = 60 }
	# 预约到期弹窗；月度检查同时兼容旧档与条件变化。
	trigger_event = { id = crb_flavor.17 months = 60 popup = yes }
}

crb_gallic_appease_france_effect = {
	if = {
		limit = { crb_can_appease_france_trigger = yes }
		# root 始终为出钱的迦太基，两国使用完全相同的一次性金额。
		add_treasury = { value = crb_gallic_appeasement_payment multiply = -1 }
		c:FRA = { add_treasury = crb_gallic_appeasement_payment }
		change_relations = { country = c:FRA value = 20 }
		create_bidirectional_truce = { country = c:FRA months = 60 }
		crb_start_gallic_policy_effect = { POLICY = crb_gallic_appeasement }
	}
}
crb_gallic_confront_france_effect = {
	if = {
		limit = { crb_can_choose_gallic_policy_trigger = yes }
		change_relations = { country = c:FRA value = -50 }
		crb_start_gallic_policy_effect = { POLICY = crb_flavor_gallic_legacy }
	}
}
crb_gallic_leave_france_effect = {
	if = {
		limit = { crb_can_choose_gallic_policy_trigger = yes }
		crb_start_gallic_policy_effect = { POLICY = crb_gallic_nonintervention }
	}
}
