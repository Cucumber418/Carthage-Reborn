# ============================================================
# 开局人口转换
# 作用域：国家（布匿）
# 首都地区 60% 的人口转为布匿文化，其余地区 20%
# 转为布匿文化的人口改信塔尼特多神教
# ============================================================
crb_convert_starting_pops_effect = {
	every_scope_state = {
		if = {
			limit = { is_capital = yes }
			every_scope_pop = {
				limit = { NOT = { has_pop_culture = crb_punic } }
				change_pop_culture = { target = cu:crb_punic value = 0.6 }
			}
		}
		else = {
			every_scope_pop = {
				limit = { NOT = { has_pop_culture = crb_punic } }
				change_pop_culture = { target = cu:crb_punic value = 0.2 }
			}
		}
	}
	every_scope_state = {
		every_scope_pop = {
			limit = {
				has_pop_culture = crb_punic
				NOT = { has_pop_religion = crb_tanit }
			}
			change_pop_religion = { target = rel:crb_tanit value = 1 }
		}
	}
}

# ============================================================
# 利益集团改名
# 作用域：国家（布匿）
# 开局执行一次，之后每月检查一次
#   田主 / 船主：未受认可时使用，受认可后变回地主 / 实业家
#   奉神者：塔尼特多神教为国教时使用
#   书记：一直使用
# 用变量记录当前状态，只在状态变化时改名
# ============================================================
crb_update_ig_names_effect = {
	# 田主、船主
	if = {
		limit = {
			is_country_type = unrecognized
			NOT = { has_variable = crb_punic_ig_names }
		}
		set_variable = crb_punic_ig_names
		every_interest_group = {
			limit = { is_interest_group_type = ig_landowners }
			set_interest_group_name = crb_ig_baal_hasadeh
		}
		every_interest_group = {
			limit = { is_interest_group_type = ig_industrialists }
			set_interest_group_name = crb_ig_baal_oniyah
		}
	}
	else_if = {
		limit = {
			NOT = { is_country_type = unrecognized }
			has_variable = crb_punic_ig_names
		}
		remove_variable = crb_punic_ig_names
		every_interest_group = {
			limit = { is_interest_group_type = ig_landowners }
			set_interest_group_name = ig_landowners
		}
		every_interest_group = {
			limit = { is_interest_group_type = ig_industrialists }
			set_interest_group_name = ig_industrialists
		}
	}

	# 奉神者
	if = {
		limit = {
			country_has_state_religion = rel:crb_tanit
			NOT = { has_variable = crb_votaries_name }
		}
		set_variable = crb_votaries_name
		every_interest_group = {
			limit = { is_interest_group_type = ig_devout }
			set_interest_group_name = crb_ig_votaries
		}
	}
	else_if = {
		limit = {
			NOT = { country_has_state_religion = rel:crb_tanit }
			has_variable = crb_votaries_name
		}
		remove_variable = crb_votaries_name
		every_interest_group = {
			limit = { is_interest_group_type = ig_devout }
			set_interest_group_name = ig_devout
		}
	}

	# 书记
	if = {
		limit = { NOT = { has_variable = crb_soperim_name } }
		set_variable = crb_soperim_name
		every_interest_group = {
			limit = { is_interest_group_type = ig_intelligentsia }
			set_interest_group_name = crb_ig_soperim
		}
	}
}

crb_barca_victory_effect = {
	# 两派胜利标记互斥，防止旧存档/残留变量误开启另一条路线
	remove_variable = crb_hanno_victory
	set_variable = crb_barca_victory

		name = crb_popular_assembly_restored
		days = long_modifier_time
	}

}

	}
	# 日志"汉诺航海记"会检查 crb_hanno_victory 变量
}

# 作用域：州
crb_assign_punic_name_to_state_effect = {
	if = {
		limit = {
			NOT = { has_game_rule = no_dynamic_naming }
			owner ?= { crb_uses_punic_names_trigger = yes }
		}
		crb_set_punic_state_name_effect = yes
	}
	else_if = {
		limit = { has_variable = crb_punic_name }
		remove_variable = crb_punic_name
		# 交还给原版的动态地名
		evaluate_and_assign_state_hub_dynamic_names = yes
	}
}

# 建立傀儡国 TRP：把迦太基在这些地区的州交给新国家（作用域：迦太基）
crb_create_puppet_TRP_effect = {
	random_scope_state = {
		limit = {
			is_capital = no
			OR = {
				state_region = s:STATE_TRIPOLI
				state_region = s:STATE_LIBYA
				state_region = s:STATE_LIBYAN_DESERT
			}
		}
		save_scope_as = crb_puppet_first_state
	}
	if = {
		limit = { exists = scope:crb_puppet_first_state }
		create_country = {
			tag = TRP
			origin = root
			state = scope:crb_puppet_first_state
			on_created = {
				set_variable = crb_carthaginian_subject
				set_state_religion = rel:crb_tanit
				add_modifier = {
					name = crb_carthaginian_protection
					days = very_long_modifier_time
				}
				crb_trading_post_laws_effect = yes
			}
		}
		every_scope_state = {
			limit = {
				is_capital = no
				OR = {
					state_region = s:STATE_TRIPOLI
					state_region = s:STATE_LIBYA
					state_region = s:STATE_LIBYAN_DESERT
				}
			}
			set_state_owner = c:TRP
		}
		if = {
