# Old-save partition: only eligible non-capital northern land of an actual GLT subject.
crb_partition_legacy_gaul_effect = {
 if = {
  limit = {
   crb_is_carthage_trigger = yes
   is_at_war = no
   crb_territory_disposition_ready_trigger = yes
   NOT = { exists = c:XLG }
   crb_is_actual_territory_subject_trigger = { TAG = GLT }
   c:GLT = {
    is_at_war = no
    any_scope_state = { crb_is_gaul_post_state_trigger = yes }
    any_scope_state = { crb_is_lugdunensis_state_trigger = yes crb_is_expansion_ready_state_trigger = yes }
   }
  }
  c:GLT = {
   every_scope_state = {
    limit = { crb_is_lugdunensis_state_trigger = yes crb_is_expansion_ready_state_trigger = yes }
    set_state_owner = c:TUN
   }
  }
  if = {
   limit = { c:GLT = { has_variable = crb_province } }
   crb_create_province_XLG_effect = yes
  }
  else = { crb_create_puppet_XLG_effect = yes }
 }
}
