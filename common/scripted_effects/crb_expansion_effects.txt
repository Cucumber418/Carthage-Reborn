# 每次只转交指定附庸的地区；最终执行时再次校验。
crb_expand_one_territory_effect = {
	if = {
		limit = { crb_can_expand_one_territory_trigger = { TAG = $TAG$ AREA = $AREA$ } }
		every_scope_state = {
			limit = { is_capital = no $AREA$ = yes }
			set_state_owner = c:$TAG$
		}
		c:$TAG$ = { crb_apply_punic_state_names_effect = yes }
	}
}
