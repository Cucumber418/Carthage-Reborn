# Country scope. Caller supplies TAG, CAN_FOUND, AREA, PROVINCE, LAWS and PACT.
crb_create_territory_subject_effect = {
	if = {
		limit = {
			is_at_war = no
			$CAN_FOUND$ = yes
		}
		save_scope_as = crb_territory_origin
		random_scope_state = {
			limit = { is_capital = no $AREA$ = yes }
			save_scope_as = crb_territory_first_state
		}
		create_country = {
			tag = $TAG$
			origin = scope:crb_territory_origin
			state = scope:crb_territory_first_state
			on_created = {
				set_variable = crb_carthaginian_subject
				set_state_religion = rel:crb_tanit
				if = {
					limit = { always = $PROVINCE$ }
					set_variable = crb_province
				}
				add_modifier = {
					name = crb_carthaginian_protection
					days = very_long_modifier_time
				}
				$LAWS$ = yes
			}
		}
		every_scope_state = {
			limit = { is_capital = no $AREA$ = yes }
			set_state_owner = c:$TAG$
		}
		if = {
			limit = { NOT = { has_modifier = crb_carthaginian_hegemony } }
			add_modifier = { name = crb_carthaginian_hegemony }
		}
		create_diplomatic_pact = { country = c:$TAG$ type = $PACT$ }
		c:$TAG$ = { crb_apply_punic_state_names_effect = yes }
	}
}

crb_try_unlock_territorial_journal_effect = {
	if = {
		limit = {
			crb_is_carthage_trigger = yes
			is_at_war = no
			NOT = { has_variable = crb_territories_unlocked }
			NOT = { has_variable = crb_territories_concluded }
			NOT = { has_journal_entry = je_crb_territories }
			crb_can_found_any_territory_trigger = yes
		}
		add_journal_entry = { type = je_crb_territories }
	}
}
crb_check_french_border_effect = {
	if = {
		limit = {
			crb_is_carthage_trigger = yes
			NOT = { has_variable = crb_flavor_17_fired }
			crb_borders_french_north_africa_trigger = yes
		}
		trigger_event = { id = crb_flavor.17 popup = yes }
	}
}
