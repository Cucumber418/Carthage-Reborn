# 迦太基的决议：扩张商站、皈依伊斯兰教
# （两千年的复仇路的各阶段改为日志里的按钮）

# 扩张商站：建立新迦太基后解锁
crb_expand_trading_posts_decision = {
	is_shown = {
		crb_is_carthage_trigger = yes
		has_variable = crb_hannibal_new_carthage_done
	}

	possible = {
		is_at_war = no
		custom_tooltip = {
			text = crb_can_expand_trading_posts_tt
			crb_can_expand_trading_posts_trigger = yes
		}
	}

	when_taken = {
		custom_tooltip = crb_expand_trading_posts_effect_tt
		hidden_effect = {
			crb_expand_trading_posts_effect = yes
		}
	}

	ai_chance = {
		value = 10
	}
}

# 皈依伊斯兰教：伊斯兰教人口超过 40% 时出现在决议面板，不弹提示
crb_convert_to_islam_decision = {
	is_shown = {
		crb_is_carthage_trigger = yes
		NOT = { country_has_state_religion = rel:sunni }
		NOT = { country_has_state_religion = rel:shiite }
		NOT = { country_has_state_religion = rel:ibadi }
		crb_islam_share > 0.4
	}

	possible = {
		custom_tooltip = {
			text = crb_islam_share_tt
			crb_islam_share > 0.4
		}
	}

	when_taken = {
		if = {
			limit = {
				religion_percent_country = { target = rel:sunni value >= 0.2 }
			}
			set_state_religion = rel:sunni
		}
		else_if = {
			limit = {
				religion_percent_country = { target = rel:shiite value >= 0.2 }
			}
			set_state_religion = rel:shiite
		}
		else = {
			set_state_religion = rel:ibadi
		}
		ruler ?= {
			change_character_religion = root.religion
		}
		heir ?= {
			change_character_religion = root.religion
		}
		# 效果参照原版国家集团的"强制改宗"，对本国使用
		capital = {
			convert_population = {
				target = root.religion
				value = 0.05
			}
		}
		every_scope_state = {
			limit = { is_capital = no }
			convert_population = {
				target = root.religion
				value = 0.02
			}
		}
		state_religion_switch_effect = yes
		add_modifier = {
			name = crb_converted_to_islam
			days = normal_modifier_time
		}
		add_radicals = {
			value = medium_radicals
			religion = rel:crb_tanit
		}
	}

	ai_chance = {
		value = 10
	}
}
